<script>
// --- CONFIGURACIÓN DE COMUNICACIÓN ---
// Nota: Para uso real, obtén una API KEY gratuita en ably.com
const ably = new Ably.Realtime('REEMPLAZA_CON_TU_API_KEY_DE_ABLY'); 
const canal = ably.channels.get('tkd-match-1');

// --- VARIABLES DE ESTADO ---
let rojo=0, azul=0, gRojo=0, gAzul=0, tiempo=120, intervalo=null, round=1;

// --- LÓGICA DE JUEZ (ENVÍO) ---
function enviarPunto(equipo, puntos) {
    canal.publish('punto', { equipo, puntos });
}

// --- LÓGICA DE MARCADOR (RECEPCIÓN) ---
canal.subscribe('punto', (msg) => {
    const data = msg.data;
    sumar(data.equipo, data.puntos);
});

// --- FUNCIONES DEL MARCADOR ---
function actualizar(){
    document.getElementById("rojo").innerText=rojo;
    document.getElementById("azul").innerText=azul;
    document.getElementById("gRojo").innerText=gRojo;
    document.getElementById("gAzul").innerText=gAzul;
    document.getElementById("tiempo").innerText=tiempo;
    document.getElementById("round").innerText=round;
}

function sumar(e,p){
    if(e==="rojo") rojo+=p;
    else azul+=p;
    actualizar();
}

function penal(e){
    if(e==="rojo"){ gRojo++; azul++; }
    else { gAzul++; rojo++; }
    actualizar();
}

function start(){
    if(intervalo) return;
    intervalo=setInterval(()=>{
        if(tiempo>0){ tiempo--; actualizar(); }
        else { pause(); alert("FIN DEL ROUND"); }
    },1000);
}

function pause(){ clearInterval(intervalo); intervalo=null; }

function reset(){
    pause();
    tiempo=120; rojo=0; azul=0; gRojo=0; gAzul=0; round++;
    actualizar();
}

function setNombres(){
    document.getElementById("nRojo").innerText=document.getElementById("rname").value||"ROJO";
    document.getElementById("nAzul").innerText=document.getElementById("aname").value||"AZUL";
}

// --- GENERAR QR ---
// El QR mandará al celular a la misma página pero con un parámetro
const urlJuez = window.location.href + "?modo=juez";
new QRCode(document.getElementById("qrcode"), {
    text: urlJuez,
    width: 128,
    height: 128
});

// Detectar si el dispositivo es Juez o Marcador
const urlParams = new URLSearchParams(window.location.search);
if (urlParams.get('modo') === 'juez') {
    document.getElementById('marcador-ui').style.display = 'none';
    document.getElementById('juez-ui').style.display = 'flex';
}
